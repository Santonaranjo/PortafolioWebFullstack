<body>
  <header class="barra">
    <div class="contenedor">
      <strong>Tienda en línea</strong>
      <a href="../index.html#proyectos">← Volver al portafolio</a>
    </div>
  </header>
  <main class="contenedor">
    <span class="etiqueta">Demo</span>
    <h1>Catálogo</h1>
    <p class="nota">Carrito: <strong id="carrito">0</strong> productos · Total: <strong id="total">$0</strong></p>
    <div class="rejilla">
      <div class="tarjeta"><h3>Audífonos</h3><p>$120.000</p><button class="boton" data-precio="120000">Agregar</button></div>
      <div class="tarjeta"><h3>Teclado</h3><p>$180.000</p><button class="boton" data-precio="180000">Agregar</button></div>
      <div class="tarjeta"><h3>Mouse</h3><p>$70.000</p><button class="boton" data-precio="70000">Agregar</button></div>
    </div>
  </main>
  <script>
    let cantidad = 0, total = 0;
    document.querySelectorAll('[data-precio]').forEach((boton) => {
      boton.addEventListener('click', () => {
        cantidad++;
        total += Number(boton.dataset.precio);
        document.getElementById('carrito').textContent = cantidad;
        document.getElementById('total').textContent = '$' + total.toLocaleString('es-CO');
      });
    });
  </script>
